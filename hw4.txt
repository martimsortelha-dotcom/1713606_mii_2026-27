#include <stdio.h>
int main() {
	int num;
	printf("Introduza um numero:");
	scanf("%d",&num);
	if(num/2==0) {
		printf("\n%%d par\n", num);
		printf("Benfica\n");
	} else {
		printf("\n%d impar\n",num);
		printf("Ice Tea\n");
	}
	printf("o numero introduzido foi %d\n,num");
	int x=23;
	int y=x;
	printf("x=%d,y=%d",x,y);
	return 0;
}
